\section*{Research Experience}

\entry{PhD Student Researcher}{Stanford Graduate School of Business}{2025 -- Present}
\begin{itemize}
  \item Built a large-scale multi-agent coordination experiment with approximately 375,000 GPT-4o trials across networks of 2--50 agents, replicating a human category-formation paradigm.
  \item Identified systematic human--AI differences in category boundaries and showed that constraining agents to human labels reduced performance by 12.7\%.
\end{itemize}

\entry{Master's Researcher}{University of Chicago}{2023 -- 2025}
\begin{itemize}
  \item Developed a psychometrically validated framework for assigning Big Five personalities to LLM agents across GPT-3.5, GPT-4, and Claude 3.
  \item Collected data from 350 participants and showed that assigned personalities predict agents' moral reasoning, risk-taking, and cooperation; published in \textit{Personality Science}.
\end{itemize}

\entry{Research Assistant}{University of British Columbia}{2020 -- 2023}
\begin{itemize}
  \item Used SEM and simulation to evaluate alternative personality scale formats and CFA fit indices; contributed to publications in \textit{Journal of Personality Assessment} and \textit{Psychological Methods}.
  \item Applied ML/NLP to 10,000+ youth essays, supervised 20 research assistants, and studied personalized AI explanations using eye-tracking; contributed to \textit{Journal of Social Issues} and an IJCAI XAI workshop paper.
